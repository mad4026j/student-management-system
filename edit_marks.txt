<?php
session_start();
require_once 'db.php';

if (!isset($_SESSION['user_id']) || $_SESSION['role'] !== 'admin') {
    header('Location: index.php');
    exit();
}

$id = isset($_GET['id']) ? intval($_GET['id']) : 0;

// Gather grade context row details along with matching student identifiers
$stmt = $conn->prepare("SELECT g.*, s.full_name, s.roll_no FROM grades g JOIN students s ON g.student_id = s.id WHERE g.id = ?");
$stmt->bind_param("i", $id);
$stmt->execute();
$grade_record = $stmt->get_result()->fetch_assoc();

if (!$grade_record) {
    die("Grade modification record index lookup target missing.");
}

$error = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $subject_name = trim($_POST['subject_name']);
    $marks_obtained = intval($_POST['marks_obtained']);
    $total_marks = intval($_POST['total_marks']);

    if (!empty($subject_name) && $marks_obtained >= 0 && $total_marks > 0) {
        $update = $conn->prepare("UPDATE grades SET subject_name = ?, marks_obtained = ?, total_marks = ? WHERE id = ?");
        $update->bind_param("siii", $subject_name, $marks_obtained, $total_marks, $id);
        
        if ($update->execute()) {
            header('Location: manage_marks.php?msg=updated');
            exit();
        } else {
            $error = "Failed to modify structural elements inside MySQL engine.";
        }
    } else {
        $error = "Parameters cannot reflect negative values or blank strings.";
    }
}

        .btn-cancel { background: #95a5a6; color: white; text-decoration: none; padding: 12px 20px; font-size: 16px; border-radius: 5px; margin-left: 10px; display: inline-block; }
        .error { color: #e74c3c; background: #fce4e4; padding: 12px; border-radius: 5px; margin-bottom: 20px; }

        <h2>Modify Score Allocation</h2>

        <form method="POST" action="">
            <div class="form-group">
                <label>Student Profile Target Context</label>
                <input type="text" value="<?php echo htmlspecialchars($grade_record['full_name'] . ' (' . $grade_record['roll_no'] . ')'); ?>" disabled style="background: #f1f2f6;">
            </div>
            <div class="form-group">
                <label>Course Paper Subject Title</label>
                <input type="text" name="subject_name" value="<?php echo htmlspecialchars($grade_record['subject_name']); ?>" required>
            </div>
            <div class="form-group">

                <input type="number" name="marks_obtained" value="<?php echo $grade_record['marks_obtained']; ?>" min="0" required>
            </div>
            <div class="form-group">
                <label>Maximum Scale Limit Score</label>
                <input type="number" name="total_marks" value="<?php echo $grade_record['total_marks']; ?>" min="1" required>
            </div>
